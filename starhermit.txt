# StarHermit engine script game: one package, played natively by the
# StarHermit client and in browsers by the engine's web runtime. Browsers open
# launch=index.html, the loader `npm run package` writes (it pins the engine
# on wiki.starhermit.com by hash; never edit it). Re-run `npm run package`
# after every change and commit index.html + starhermit-files.json.
launch=index.html
engine.api=1
engine.entry=native/main.ts
engine.version=1.29.0
engine.minor=31
engine.capabilities=platform,storage,clipboard

# Keyboard controls (players rebind keys and pads in the runtime's Options).
# The page is also drawn on with the mouse or a finger (drag), and the
# tray's buttons do Release / Undo / Hint / Skip / Pause.
control.pen_left=ArrowLeft | Move pen left
control.pen_right=ArrowRight | Move pen right
control.pen_up=ArrowUp | Move pen up
control.pen_down=ArrowDown | Move pen down
control.pen_fast=ShiftLeft+ShiftRight | Move pen faster (hold)
control.pen_toggle=Space | Pen down / pen up
control.release=KeyR | Release the storm
control.undo=KeyU | Undo last stroke
control.hint=KeyH | Hint
control.skip=KeyS | Skip storm playback
control.pause=KeyP+Escape | Pause / resume
control.camera=KeyC | Reset camera framing
engine.group.pen_left=Pen
engine.group.release=Round

# Gamepad defaults: the D-pad or left stick moves the pen (the left shoulder
# speeds it up), A puts the pen down and lifts it (drawing the stroke), Y
# releases the storm, X undoes, Select asks for a hint, the right shoulder
# skips the storm, Start pauses.
engine.gamepad.pen_left=DpadLeft+LeftStickLeft
engine.gamepad.pen_right=DpadRight+LeftStickRight
engine.gamepad.pen_up=DpadUp+LeftStickUp
engine.gamepad.pen_down=DpadDown+LeftStickDown
engine.gamepad.pen_fast=LeftShoulder
engine.gamepad.pen_toggle=South
engine.gamepad.release=North
engine.gamepad.undo=West
engine.gamepad.hint=Select
engine.gamepad.skip=RightShoulder
engine.gamepad.pause=Start
